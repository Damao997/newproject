## 二、docker-compose.yml（完整配置）

```yaml
# docker-compose.yml
# 浙江壹品慧财年经营数据分析平台 — Docker 单机部署

version: '3.8'

services:
  # ═══════════════════════════════════════════════════════
  # PostgreSQL 数据库
  # ═══════════════════════════════════════════════════════
  postgres:
    image: postgres:15-alpine
    container_name: yipinhui-postgres
    restart: unless-stopped
    environment:
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: ${DB_SUPER_PASSWORD}
      POSTGRES_DB: yipinhui
    volumes:
      - pg_data:/var/lib/postgresql/data
      - ./init-roles.sql:/docker-entrypoint-initdb.d/01-init-roles.sql
    ports:
      - "127.0.0.1:5432:5432"     # 仅本地访问，不暴露到公网
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres -d yipinhui"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 30s
    networks:
      - app-network
    # PostgreSQL 配置（备份相关）
    command: |
      postgres
      -c archive_mode=on
      -c archive_command='test ! -f /backup_data/wal/%f && cp %p /backup_data/wal/%f'
      -c wal_level=replica
      -c max_wal_senders=3

  # ═══════════════════════════════════════════════════════
  # 后端 API 服务
  # ═══════════════════════════════════════════════════════
  backend:
    build:
      context: ./server
      dockerfile: Dockerfile
    container_name: yipinhui-backend
    restart: unless-stopped
    environment:
      NODE_ENV: production
      PORT: 3001
      # 数据库连接（使用 yipinhui_app 业务账号）
      DATABASE_URL: postgresql://yipinhui_app:${DB_APP_PASSWORD}@postgres:5432/yipinhui
      # JWT 密钥
      JWT_SECRET: ${JWT_SECRET}
      JWT_REFRESH_SECRET: ${JWT_REFRESH_SECRET}
      # DeepSeek AI API
      DEEPSEEK_API_KEY: ${DEEPSEEK_API_KEY}
      DEEPSEEK_API_BASE: https://api.deepseek.com/v1
      # 前端 Origin（用于 CORS 白名单）
      FRONTEND_ORIGIN: ${FRONTEND_ORIGIN:-http://localhost}
    ports:
      - "127.0.0.1:3001:3001"
    depends_on:
      postgres:
        condition: service_healthy
    networks:
      - app-network
    healthcheck:
      test: ["CMD", "wget", "--no-verbose", "--tries=1", "--spider", "http://localhost:3001/api/v1/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 15s

  # ═══════════════════════════════════════════════════════
  # 前端 Web 服务（Nginx 静态文件 + API 反向代理）
  # ═══════════════════════════════════════════════════════
  frontend:
    build:
      context: ./web
      dockerfile: Dockerfile
    container_name: yipinhui-frontend
    restart: unless-stopped
    ports:
      - "80:80"
    depends_on:
      - backend
    networks:
      - app-network
    healthcheck:
      test: ["CMD", "wget", "--no-verbose", "--tries=1", "--spider", "http://localhost/"]
      interval: 30s
      timeout: 5s
      retries: 3

  # ═══════════════════════════════════════════════════════
  # 数据库备份服务（定时 cron）
  # ═══════════════════════════════════════════════════════
  db-backup:
    image: postgres:15-alpine
    container_name: yipinhui-backup
    restart: unless-stopped
    environment:
      PGHOST: postgres
      PGUSER: postgres
      PGPASSWORD: ${DB_SUPER_PASSWORD}
      PGDATABASE: yipinhui
      BACKUP_ENCRYPTION_KEY: ${BACKUP_ENCRYPTION_KEY}
    volumes:
      - backup_data:/backup_data
      - ./backup.sh:/usr/local/bin/backup.sh:ro
    entrypoint: ["/bin/sh", "-c"]
    command: |
      "
      # 创建 crontab: 每日凌晨 2:00 执行全量备份
      echo '0 2 * * * /usr/local/bin/backup.sh full >> /backup_data/backup.log 2>&1' > /etc/crontabs/root;
      # 每小时 WAL 归档轮转
      echo '0 * * * * /usr/local/bin/backup.sh rotate >> /backup_data/backup.log 2>&1' >> /etc/crontabs/root;
      # 启动 crond 前台运行
      exec crond -f -l 2
      "
    depends_on:
      postgres:
        condition: service_healthy
    networks:
      - app-network

# ═══════════════════════════════════════════════════════
# Volumes
# ═══════════════════════════════════════════════════════
volumes:
  pg_data:
    driver: local
  backup_data:
    driver: local

# ═══════════════════════════════════════════════════════
# Networks
# ═══════════════════════════════════════════════════════
networks:
  app-network:
    driver: bridge
```

> **注意**: DATABASE_URL 使用业务账号 `yipinhui_app`（非超级用户）连接数据库。Prisma migration 由超级用户手动执行。

## 三、Nginx 配置

### 3.1 完整 nginx.conf

```nginx
# /etc/nginx/nginx.conf
# 浙江壹品慧财年经营数据分析平台 — Nginx 反向代理配置

worker_processes auto;
error_log /var/log/nginx/error.log warn;
pid /var/run/nginx.pid;

events {
    worker_connections 1024;
    use epoll;
}

http {
    include       /etc/nginx/mime.types;
    default_type  application/octet-stream;

    # 日志格式
    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                    '$status $body_bytes_sent "$http_referer" '
                    '"$http_user_agent" "$http_x_forwarded_for"';

    access_log /var/log/nginx/access.log main;

    # 性能优化
    sendfile        on;
    tcp_nopush      on;
    tcp_nodelay     on;
    keepalive_timeout 65;
    types_hash_max_size 2048;

    # Gzip 压缩
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_comp_level 6;
    gzip_types
        text/plain
        text/css
        text/javascript
        application/javascript
        application/json
        application/xml
        image/svg+xml;

    # ═══════════════════════════════════════════════════════
    # 上游后端服务
    # ═══════════════════════════════════════════════════════
    upstream backend {
        server backend:3001;
        keepalive 32;
    }

    server {
        listen 80;
        server_name localhost;

        # 客户端上传大小限制
        client_max_body_size 50M;

        # ═══════════════════════════════════════════════════
        # HTTP 安全头
        # ═══════════════════════════════════════════════════
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-XSS-Protection "1; mode=block" always;
        add_header Referrer-Policy "strict-origin-when-cross-origin" always;
        add_header Permissions-Policy "camera=(), microphone=(), geolocation=()" always;

        # ═══════════════════════════════════════════════════
        # 前端静态文件
        # ═══════════════════════════════════════════════════
        root /usr/share/nginx/html;
        index index.html;

        # SPA 路由回退: 非 /api/ 路径返回 index.html
        location / {
            try_files $uri $uri/ /index.html;

            # 静态资源缓存
            location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
                expires 30d;
                add_header Cache-Control "public, immutable";
            }
        }

        # ═══════════════════════════════════════════════════
        # API 反向代理（通用路由）
        # ═══════════════════════════════════════════════════
        location /api/ {
            proxy_pass http://backend;
            proxy_http_version 1.1;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_set_header Connection "";

            # 超时配置
            proxy_connect_timeout 30s;
            proxy_send_timeout 60s;
            proxy_read_timeout 60s;

            # 缓冲配置
            proxy_buffering on;
            proxy_buffer_size 4k;
            proxy_buffers 8 4k;
            proxy_busy_buffers_size 8k;
        }

        # ═══════════════════════════════════════════════════
        # SSE 流式路由（AI 润色/分析）
        # 禁用缓冲以支持实时流式传输
        # ═══════════════════════════════════════════════════
        location /api/v1/ai/ {
            proxy_pass http://backend;
            proxy_http_version 1.1;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # ⚠️ SSE 必须禁用缓冲
            proxy_buffering off;
            proxy_cache off;

            # SSE 长连接超时（5 分钟）
            proxy_read_timeout 300s;
            proxy_send_timeout 300s;

            # Chunked Transfer Encoding
            chunked_transfer_encoding on;

            # 禁用 nginx 缓冲响应
            proxy_request_buffering off;

            # 告诉上游不要缓冲（Express 端也会设置）
            proxy_set_header X-Accel-Buffering no;
        }

        # ═══════════════════════════════════════════════════
        # 文件上传路由（支持大文件）
        # ═══════════════════════════════════════════════════
        location /api/v1/data/imports {
            proxy_pass http://backend;
            proxy_http_version 1.1;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # 上传大文件超时
            proxy_connect_timeout 60s;
            proxy_send_timeout 120s;
            proxy_read_timeout 120s;
        }

        # ═══════════════════════════════════════════════════
        # 健康检查端点
        # ═══════════════════════════════════════════════════
        location /api/v1/health {
            proxy_pass http://backend;
            proxy_http_version 1.1;
            proxy_set_header Host $host;
            access_log off;  # 健康检查不记录访问日志
        }

        # ═══════════════════════════════════════════════════
        # 安全策略: 禁止访问隐藏文件
        # ═══════════════════════════════════════════════════
        location ~ /\. {
            deny all;
            access_log off;
            log_not_found off;
        }
    }
}
```

**SSE 关键配置说明**:
- `proxy_buffering off`：必须，否则 nginx 默认缓冲导致 SSE 流式失效
- `proxy_cache off`：必须，防止缓存流式响应
- `proxy_read_timeout 300s`：长连接超时，AI 生成可能耗时较长
- `chunked_transfer_encoding on`：支持分块传输
- `X-Accel-Buffering: no`：告知客户端不缓冲

**Express 端 SSE 路由注意事项**:
- SSE 路由禁用 `compression` 中间件
- 响应头：`Content-Type: text/event-stream`、`Cache-Control: no-cache`、`Connection: keep-alive`

## 四、数据库初始化

### 4.1 init-roles.sql（完整实现）

```sql
-- ============================================================
-- init-roles.sql
-- 浙江壹品慧财年经营数据分析平台 — 数据库初始化脚本
-- 执行时机: PostgreSQL 容器首次启动时自动执行
-- 挂载路径: /docker-entrypoint-initdb.d/01-init-roles.sql
-- ============================================================

-- 1. 创建业务账号 yipinhui_app
DO $$
BEGIN
  IF NOT EXISTS (SELECT FROM pg_catalog.pg_roles WHERE rolname = 'yipinhui_app') THEN
    CREATE ROLE yipinhui_app WITH LOGIN PASSWORD :'DB_APP_PASSWORD';
    RAISE NOTICE '业务账号 yipinhui_app 创建成功';
  ELSE
    RAISE NOTICE '业务账号 yipinhui_app 已存在，跳过创建';
  END IF;
END $$;

-- 2. 授权数据库连接
GRANT CONNECT ON DATABASE yipinhui TO yipinhui_app;

-- 3. 授权 schema 使用
GRANT USAGE ON SCHEMA public TO yipinhui_app;

-- 4. 对所有现有表授予 DML 权限（SELECT/INSERT/UPDATE/DELETE）
--    不授予 DDL 权限（CREATE/ALTER/DROP/TRUNCATE）
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA public TO yipinhui_app;

-- 5. 对所有现有序列授予使用权限
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO yipinhui_app;

-- 6. 对所有现有函数授予执行权限
GRANT EXECUTE ON ALL FUNCTIONS IN SCHEMA public TO yipinhui_app;

-- 7. audit_log 特殊权限: INSERT + SELECT only（不可 UPDATE/DELETE，防篡改）
--    注意: 此 REVOKE 必须在 audit_log 表创建后执行
--    如果 audit_log 表尚未创建（由 Prisma migration 创建），以下语句将在后续执行
DO $$
BEGIN
  IF EXISTS (SELECT FROM information_schema.tables WHERE table_name = 'audit_log') THEN
    REVOKE UPDATE, DELETE ON audit_log FROM yipinhui_app;
    RAISE NOTICE 'audit_log 防篡改权限设置完成 (仅 INSERT + SELECT)';
  ELSE
    RAISE NOTICE 'audit_log 表尚未创建，防篡改权限待 Prisma migration 后执行';
  END IF;
END $$;

-- 8. token_blacklist 特殊权限: INSERT + SELECT only
DO $$
BEGIN
  IF EXISTS (SELECT FROM information_schema.tables WHERE table_name = 'token_blacklist') THEN
    REVOKE UPDATE, DELETE ON token_blacklist FROM yipinhui_app;
    RAISE NOTICE 'token_blacklist 权限设置完成 (仅 INSERT + SELECT)';
  ELSE
    RAISE NOTICE 'token_blacklist 表尚未创建，权限待 Prisma migration 后执行';
  END IF;
END $$;

-- 9. 设置默认权限（对将来创建的表自动生效）
ALTER DEFAULT PRIVILEGES IN SCHEMA public
  GRANT SELECT, INSERT, UPDATE, DELETE ON TABLES TO yipinhui_app;

ALTER DEFAULT PRIVILEGES IN SCHEMA public
  GRANT USAGE, SELECT ON SEQUENCES TO yipinhui_app;

ALTER DEFAULT PRIVILEGES IN SCHEMA public
  GRANT EXECUTE ON FUNCTIONS TO yipinhui_app;

-- 10. 验证权限配置
SELECT
  table_name,
  string_agg(privilege_type, ', ' ORDER BY privilege_type) AS granted_privileges
FROM information_schema.table_privileges
WHERE grantee = 'yipinhui_app'
  AND table_schema = 'public'
GROUP BY table_name
ORDER BY table_name;

RAISE NOTICE '========================================';
RAISE NOTICE '数据库初始化完成';
RAISE NOTICE '业务账号: yipinhui_app (DML only)';
RAISE NOTICE '========================================';
```

### 4.2 数据库账号权限矩阵

| 账号 | 用途 | 权限 |
|------|------|------|
| `postgres`（超级用户） | 仅 Prisma migration 初始化 | 全部（DDL + DML） |
| `yipinhui_app`（业务账号） | 应用连接 | **仅 DML**（SELECT/INSERT/UPDATE/DELETE），不授予 DDL |
| `yipinhui_app` 对 `audit_log` | 审计日志 | **仅 INSERT + SELECT**（不可 UPDATE/DELETE，不可篡改） |
| `yipinhui_app` 对 `token_blacklist` | Token 黑名单 | **仅 INSERT + SELECT**（不可 UPDATE/DELETE） |

### 4.3 Prisma Migration 后补充权限脚本

```sql
-- post-migration-fix-permissions.sql
-- 在每次 Prisma migration 后执行，确保新创建的表权限正确

-- 对新表授予 DML
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA public TO yipinhui_app;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO yipinhui_app;

-- audit_log 防篡改（每次 migration 后可能被重置）
REVOKE UPDATE, DELETE ON audit_log FROM yipinhui_app;

-- token_blacklist 防篡改
REVOKE UPDATE, DELETE ON token_blacklist FROM yipinhui_app;
```

| 备份加密 | `openssl enc -aes-256-cbc` |
| 全量保留期 | 30 天 |
| WAL 保留期 | 7 天 |
| 备份存储 | 独立 Volume `backup_data` |
| 恢复演练 | 每季度一次 |

### 6.2 backup.sh 完整脚本

```bash
#!/bin/sh
# backup.sh
# 浙江壹品慧财年经营数据分析平台 — 数据库备份脚本
# 挂载路径: /usr/local/bin/backup.sh
# 由 db-backup 容器的 crond 定时调用

set -e

BACKUP_DIR="/backup_data"
FULL_DIR="${BACKUP_DIR}/full"
WAL_DIR="${BACKUP_DIR}/wal"
LOG_FILE="${BACKUP_DIR}/backup.log"
ENCRYPTION_KEY="${BACKUP_ENCRYPTION_KEY:-change_me}"

log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a "${LOG_FILE}"
}

# ═══════════════════════════════════════════════════════
# 全量备份
# ═══════════════════════════════════════════════════════
do_full_backup() {
    TIMESTAMP=$(date '+%Y%m%d_%H%M%S')
    BACKUP_FILE="${FULL_DIR}/yipinhui_full_${TIMESTAMP}.sql"
    ENCRYPTED_FILE="${BACKUP_FILE}.enc"

    log "开始全量备份..."

    mkdir -p "${FULL_DIR}"

    # 执行 pg_dump
    PGPASSWORD="${PGPASSWORD}" pg_dump \
        -h "${PGHOST:-postgres}" \
        -U "${PGUSER:-postgres}" \
        -d "${PGDATABASE:-yipinhui}" \
        -F c \
        -v \
        -f "${BACKUP_FILE}" 2>&1 | tee -a "${LOG_FILE}"

    if [ $? -eq 0 ]; then
        log "pg_dump 完成: ${BACKUP_FILE}"

        # AES-256-CBC 加密
        openssl enc -aes-256-cbc \
            -salt \
            -pbkdf2 \
            -pass pass:"${ENCRYPTION_KEY}" \
            -in "${BACKUP_FILE}" \
            -out "${ENCRYPTED_FILE}"

        if [ $? -eq 0 ]; then
            log "加密完成: ${ENCRYPTED_FILE}"
            # 加密成功后删除原始文件
            rm -f "${BACKUP_FILE}"
        else
            log "加密失败，保留原始备份文件"
        fi
    else
        log "全量备份失败!"
        return 1
    fi

    log "全量备份结束"
}

# ═══════════════════════════════════════════════════════
# 清理过期备份
# ═══════════════════════════════════════════════════════
do_rotate() {
    log "开始备份轮转..."

    # 清理 30 天前的全量备份
    if [ -d "${FULL_DIR}" ]; then
        DELETED_FULL=$(find "${FULL_DIR}" -name "*.enc" -type f -mtime +30 -delete -print | wc -l)
        log "清理全量备份: ${DELETED_FULL} 个文件 (保留 30 天)"
    fi

    # 清理 7 天前的 WAL 归档
    if [ -d "${WAL_DIR}" ]; then
        DELETED_WAL=$(find "${WAL_DIR}" -type f -mtime +7 -delete -print | wc -l)
        log "清理 WAL 归档: ${DELETED_WAL} 个文件 (保留 7 天)"
    fi

    log "备份轮转结束"
}

# ═══════════════════════════════════════════════════════
# 恢复测试（每季度手动执行）
# ═══════════════════════════════════════════════════════
do_restore_test() {
    log "开始恢复演练..."

    LATEST_BACKUP=$(ls -t "${FULL_DIR}"/*.enc 2>/dev/null | head -1)
    if [ -z "${LATEST_BACKUP}" ]; then
        log "无可用备份，演练取消"
        return 1
    fi

    DECRYPTED_FILE="${LATEST_BACKUP%.enc}.restore_test.sql"
    log "解密测试: ${LATEST_BACKUP}"

    # 解密
    openssl enc -aes-256-cbc -d \
        -pbkdf2 \
        -pass pass:"${ENCRYPTION_KEY}" \
        -in "${LATEST_BACKUP}" \
        -out "${DECRYPTED_FILE}" 2>&1

    if [ $? -eq 0 ]; then
        log "解密成功: ${DECRYPTED_FILE}"

        # 验证文件完整性（检查文件大小 > 0）
        SIZE=$(stat -c%s "${DECRYPTED_FILE}" 2>/dev/null || echo 0)
        if [ "${SIZE}" -gt 0 ]; then
            log "备份文件完整性验证通过 (大小: ${SIZE} bytes)"
        else
            log "备份文件大小为 0，可能损坏!"
            return 1
        fi

        # 清理测试文件
        rm -f "${DECRYPTED_FILE}"
    else
        log "解密失败!"
        return 1
    fi

    log "恢复演练结束"
}

# ═══════════════════════════════════════════════════════
# 主入口
# ═══════════════════════════════════════════════════════
case "${1}" in
    full)
        do_full_backup
        ;;
    rotate)
        do_rotate
        ;;
    restore-test)
        do_restore_test
        ;;
    *)
        echo "用法: $0 {full|rotate|restore-test}"
        echo "  full         - 执行全量备份 + 加密"
        echo "  rotate       - 清理过期备份 (30天全量 / 7天WAL)"
        echo "  restore-test - 恢复演练 (解密最新备份并验证完整性)"
        exit 1
        ;;
esac
```

### 6.3 恢复演练清单

每季度执行一次恢复演练，记录结果：

| # | 步骤 | 命令 | 预期结果 |
|---|------|------|----------|
| 1 | 列出可用备份 | `ls -la /backup_data/full/` | 最近 30 天内有备份文件 |
| 2 | 解密最新备份 | `docker exec yipinhui-backup backup.sh restore-test` | 解密成功，完整性验证通过 |
| 3 | 创建恢复测试库 | `createdb -U postgres yipinhui_restore_test` | 数据库创建成功 |
| 4 | 导入备份 | `pg_restore -d yipinhui_restore_test <backup_file>` | 数据导入成功 |
| 5 | 验证关键表 | `SELECT count(*) FROM company;` 等 | 行数与生产一致 |
| 6 | 清理测试库 | `dropdb -U postgres yipinhui_restore_test` | 清理成功 |

```typescript
// routes/health.ts

import { Router } from 'express';

const router = Router();
```

### 7.3 PostgreSQL 监控查询

```sql
-- 数据库连接数
SELECT count(*) AS active_connections FROM pg_stat_activity WHERE state = 'active';

-- 数据库大小
SELECT pg_database_size('yipinhui') / 1024 / 1024 AS size_mb;

-- 表大小（Top 10）
SELECT
  tablename,
  pg_size_pretty(pg_total_relation_size(schemaname||'.'||tablename)) AS size
FROM pg_tables
WHERE schemaname = 'public'
ORDER BY pg_total_relation_size(schemaname||'.'||tablename) DESC
LIMIT 10;

-- 慢查询（> 1 秒）
SELECT
  pid,
  now() - pg_stat_activity.query_start AS duration,
  query,
  state
FROM pg_stat_activity
WHERE (now() - pg_stat_activity.query_start) > interval '1 seconds'
  AND state = 'active'
ORDER BY duration DESC;
```

## 九、CSS 动画性能注意事项

v3 方案使用纯 CSS Animation（transition + @keyframes），无 backdrop-filter 毛玻璃效果，性能开销极低，无需特殊降级策略。确保 CSS 动画使用 transform 和 opacity（GPU 加速属性），避免在动画中使用 layout-triggering 属性（width/height/top/left）。
